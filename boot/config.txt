# Raspberry Pi firmware configuration for Dazzle.
# https://www.raspberrypi.com/documentation/computers/config_txt.html

# The system is in one of two OS slots, slot-a/ and slot-b/: the firmware
# looks for the kernel, the initramfs, cmdline.txt, the device trees and
# (as the slot's overlays/ has a README) the overlays there. (An update
# writes the other slot, then switches this.)
os_prefix=slot-a/

# The firmware picks the kernel (kernel.img on the ARMv6 boards: the Pi 1,
# Zero and Zero W; kernel7.img on the Pi 2, 3, Zero 2 W and 4; kernel8.img
# on the Pi 5), and the board's device tree, from the slot by itself.
arm_64bit=0
[pi4]
kernel=kernel7.img
[pi5]
arm_64bit=1
kernel=kernel8.img
[all]
initramfs initramfs.cpio followkernel

# The kernel's vc4 driver drives the display and the GPU (it is off in the
# device trees until this overlay turns it on), and dtparam=audio=on enables
# the onboard 3.5 mm headphone jack alongside HDMI audio.
dtoverlay=vc4-kms-v3d
dtparam=audio=on

# The camera: the firmware finds which Camera Module is on the connector
# and applies its overlay from the slot's overlays/ (imx219 or ov5647, the
# Raspberry Pi kernel's). To choose one by hand instead, set
# camera_auto_detect=0 and add dtoverlay=imx219 (or ov5647).
camera_auto_detect=1
# The firmware would otherwise look for the official touchscreen on the
# I2C pins the camera uses (on some boards), which Linux drives.
disable_touchscreen=1

# The I2C bus on the GPIO header (pins 3 and 5: GPIO 2 and 3, which have
# pull-up resistors on the board for it), for sensors and converters the
# Data Logger reads.
dtparam=i2c_arm=on

# Serial console on GPIO 14 (TX) / 15 (RX), 115200 8N1. On the boards with
# Bluetooth this is the mini UART, which needs a fixed core clock.
enable_uart=1

# Run the CPU and GPU at full speed all the time. Without this the firmware
# runs at full speed only briefly after boot, then drops to its minimum
# clocks (CPU 700 MHz, GPU core 250 MHz on a Zero) until a kernel cpufreq
# driver asks for more, and Dazzle has none: the demo fell from 60 to
# 30 fps after about a minute. The firmware still slows down if the SoC
# overheats or the supply voltage sags. (Without over_voltage this does not
# affect the warranty.)
force_turbo=1

# The kernel's vc4 driver owns the display. Keep the firmware from drawing
# its rainbow splash or warning overlays on top of it.
disable_splash=1
avoid_warnings=2
