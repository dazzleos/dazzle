# Raspberry Pi firmware configuration for Dazzle.
# https://www.raspberrypi.com/documentation/computers/config_txt.html

# The system is in one of two OS slots, slot-a/ and slot-b/: the firmware
# looks for the kernel, the initramfs, cmdline.txt and the device trees
# there. (An update writes the other slot, then switches this.)
os_prefix=slot-a/

# Mainline kernel and device tree; the firmware's default names are for the
# Raspberry Pi kernel fork. [pi0] also matches the Zero W, so it comes first.
kernel=zImage
initramfs initramfs.cpio followkernel

[pi0]
device_tree=bcm2835-rpi-zero.dtb

[pi0w]
device_tree=bcm2835-rpi-zero-w.dtb

# Pi 3 A+ (and, wrongly, the 3 B+, which is not supported). The firmware
# boots it in 32-bit mode, as Dazzle's kernel needs.
[pi3+]
device_tree=bcm2837-rpi-3-a-plus.dtb

[all]
# Serial console on GPIO 14 (TX) / 15 (RX), 115200 8N1. On the Zero W and
# Pi 3 A+ this is the mini UART, which needs a fixed core clock.
enable_uart=1

# Run the CPU and GPU at full speed all the time. Without this the firmware
# runs at full speed only briefly after boot, then drops to its minimum
# clocks (CPU 700 MHz, GPU core 250 MHz on a Zero) until a kernel cpufreq
# driver asks for more, and Dazzle has none: the demo fell from 60 to
# 30 fps after about a minute. The firmware still slows down if the SoC
# overheats or the supply voltage sags. (Without over_voltage this does not
# affect the warranty.)
force_turbo=1

# The kernel's vc4 driver owns the display. Keep the firmware from drawing
# its rainbow splash or warning overlays on top of it.
disable_splash=1
avoid_warnings=2
